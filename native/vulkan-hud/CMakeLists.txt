cmake_minimum_required(VERSION 3.16)
project(jacob_vulkan_hud LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_path(VULKAN_INCLUDE_DIR vulkan/vulkan.h REQUIRED)
add_library(VK_LAYER_JACOB_hud SHARED jacob_vk_layer.cpp)
target_include_directories(VK_LAYER_JACOB_hud PRIVATE "${VULKAN_INCLUDE_DIR}")
target_compile_options(VK_LAYER_JACOB_hud PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden)
set_target_properties(VK_LAYER_JACOB_hud PROPERTIES OUTPUT_NAME "VK_LAYER_JACOB_hud" PREFIX "lib")
